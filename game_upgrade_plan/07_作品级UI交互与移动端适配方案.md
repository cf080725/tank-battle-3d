## 🛠️ 底层数据结构设计

```javascript
/* 移动端虚拟双摇杆 */
const touchControls = {
  isTouchDevice: ('ontouchstart' in window) || (navigator.maxTouchPoints > 0),
  leftJoy: { active: false, startX: 0, startY: 0, currentX: 0, currentY: 0, dirX: 0, dirY: 0 },
  rightJoy: { active: false, startX: 0, startY: 0, currentX: 0, currentY: 0, aimAngle: 0, isFiring: false }
};

/* 粒子与对象池 (Object Pool for Zero Garbage Collection) */
class ParticlePool {
  constructor(size = 300) {
    this.pool = [];
    for (let i = 0; i < size; i++) {
      const p = new THREE.Mesh(sparkGeom, sparkMat);
      p.visible = false;
      scene.add(p);
      this.pool.push({ mesh: p, inUse: false, vx: 0, vy: 0, vz: 0, life: 0 });
    }
  }
  
  get() {
    return this.pool.find(p => !p.inUse);
  }
}
```
